// Optional represents an object property that may be absent. The zero value is
// absent; a present value may contain T's zero value but may not encode as null.
// Containing struct fields must use json:",omitzero" so absent values are
// omitted before MarshalJSON is called.
type Optional[T any] struct {
	Present bool
	Value   T
}

// IsZero reports whether the property is absent.
func (o Optional[T]) IsZero() bool { return !o.Present }

// MarshalJSON encodes a present non-null value.
func (o Optional[T]) MarshalJSON() ([]byte, error) {
	if !o.Present {
		return nil, errOptionalAbsent
	}
	return marshalPresent(o.Value)
}

// UnmarshalJSON decodes a present non-null value without mutating the receiver
// when decoding fails.
func (o *Optional[T]) UnmarshalJSON(data []byte) error {
	if isJSONNull(data) {
		return errors.New("Optional value cannot be JSON null")
	}
	var value T
	if err := json.Unmarshal(data, &value); err != nil {
		return err
	}
	*o = Optional[T]{Present: true, Value: value}
	return nil
}

func (e *emitter) decodeField(field typegrammar.Field, object, out, at string) error {
	target := out + "." + field.GoName
	key := strconv.Quote(field.JSONName)
	fieldPath := e.childPath(at, field.JSONName)

	required := func() string {
		value := e.name("raw")
		e.writef("%s, err := dvRequired(%s, %s, %s)", value, object, key, fieldPath)
		e.check()
		return value
	}
	optional := func() string {
		value, present := e.name("raw"), e.name("ok")
		e.writef("%s, %s, err := dvPresent(%s, %s, %s)", value, present, object, key, fieldPath)
		e.check()
		e.writef("if %s {", present)
		return value
	}

	switch n := field.Value.(type) {
	case *typegrammar.Required:
		return e.assign(n.Type, target, required(), fieldPath)

	case *typegrammar.Optional:
		value := optional()
		e.writef("%s.Present = true", target)
		if err := e.assign(n.Type, target+".Value", value, fieldPath); err != nil {
			return err
		}
		e.writef("}")
		return nil

func dvBool(raw any, at string) (bool, error) {
	v, ok := raw.(bool)
	if !ok {
		return false, dvErr(at, "expected a boolean, got %s", dvKind(raw))
	}
	return v, nil
}

func dvString(raw any, at string) (string, error) {
	v, ok := raw.(string)
	if !ok {
		return "", dvErr(at, "expected a string, got %s", dvKind(raw))
	}
	return v, nil
}

func dvNumber(raw any, at string) (float64, error) {
	v, ok := raw.(float64)
	if !ok {
		return 0, dvErr(at, "expected a number, got %s", dvKind(raw))
	}
	return v, nil
}

func dvInteger(raw any, at string, min, max float64) (float64, error) {
	v, err := dvNumber(raw, at)
	if err != nil {
		return 0, err
	}
	if math.IsInf(v, 0) || math.IsNaN(v) || v != math.Trunc(v) {
		return 0, dvErr(at, "expected an integer, got %v", v)
	}
	if v < min || v > max {
		return 0, dvErr(at, "number %v is out of range", v)
	}
	return v, nil
}

// dvPresent reads an optional property. It reports absence, and rejects an
// explicit undefined, which is not the same thing on this wire.
func dvPresent(o *devalue.Object, key string, at string) (any, bool, error) {
	v, ok := o.Get(key)
	if !ok {
		return nil, false, nil
	}
	if _, undefined := v.(devalue.UndefinedValue); undefined {
		return nil, false, dvErr(at, "value is undefined")
	}
	return v, true, nil
}
